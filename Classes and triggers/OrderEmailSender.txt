public without sharing class OrderEmailSender {

    @TestVisible
    private static Integer emailsPrepared = 0;

    @TestVisible
    private static Integer emailsSent = 0;

    public static void sendOrderEmail(Set<Id> orderIds) {

        emailsPrepared = 0;
        emailsSent = 0;

        if (orderIds == null || orderIds.isEmpty()) {
            return;
        }

        // Get Orders
        List<AgriEdge_Order__c> orders = [
            SELECT Id,
                   Name,
                   Total_Amount__c,
                   Payment_Status__c,
                   Order_Status__c,
                   Customer__c,
                   CreatedDate,
                   Shipping_Address__c,
                   Discounted_Total__c
            FROM AgriEdge_Order__c
            WHERE Id IN :orderIds
        ];

        if (orders.isEmpty()) {
            return;
        }

        // Get Customer Account IDs
        Set<Id> accountIds = new Set<Id>();

        for (AgriEdge_Order__c order : orders) {
            if (order.Customer__c != null) {
                accountIds.add(order.Customer__c);
            }
        }

        if (accountIds.isEmpty()) {
            return;
        }

        // Get Contacts
        Map<Id, List<String>> accountEmails =
            new Map<Id, List<String>>();

        for (Contact contact : [
            SELECT Id, Email, AccountId
            FROM Contact
            WHERE AccountId IN :accountIds
            AND Email != null
        ]) {

            if (!accountEmails.containsKey(contact.AccountId)) {
                accountEmails.put(
                    contact.AccountId,
                    new List<String>()
                );
            }

            accountEmails.get(contact.AccountId).add(contact.Email);
        }

        // Create emails
        List<Messaging.SingleEmailMessage> emails =
            new List<Messaging.SingleEmailMessage>();

        for (AgriEdge_Order__c order : orders) {

            if (order.Customer__c == null) {
                continue;
            }

            if (!accountEmails.containsKey(order.Customer__c)) {
                continue;
            }

            List<String> toEmails =
                accountEmails.get(order.Customer__c);

            if (toEmails == null || toEmails.isEmpty()) {
                continue;
            }

            String emailBody =
                'Dear Customer,<br/><br/>' +
                'Your order has been updated with the following details:<br/><br/>' +

                '<b>Order Name:</b> ' +
                order.Name + '<br/>' +

                '<b>Order Status:</b> ' +
                order.Order_Status__c + '<br/>' +

                '<b>Total Amount:</b> ' +
                order.Total_Amount__c + '<br/>' +

                '<b>Payment Status:</b> ' +
                order.Payment_Status__c + '<br/>' +

                '<b>Shipping Address:</b> ' +
                order.Shipping_Address__c + '<br/>' +

                '<b>Created Date:</b> ' +
                order.CreatedDate + '<br/><br/>' +

                '<b>Total Amount Paid (Including Discount):</b> ' +
                order.Discounted_Total__c + '<br/><br/>' +

                'Thank you for your business!<br/><br/>' +
                'Best Regards,<br/>' +
                'Your Company Name';

            Messaging.SingleEmailMessage email =
                new Messaging.SingleEmailMessage();

            email.setToAddresses(toEmails);
            email.setSubject(
                'Your Order Payment Status has been Updated'
            );
            email.setHtmlBody(emailBody);

            emails.add(email);
        }

        // Store how many emails were prepared
        emailsPrepared = emails.size();

        // Send
        if (!emails.isEmpty()) {

            Messaging.SendEmailResult[] results =
                Messaging.sendEmail(emails);

            for (Messaging.SendEmailResult result : results) {

                if (result.isSuccess()) {
                    emailsSent++;
                }
            }
        }
    }
}
